   347→   347→function ProfileCard({
   348→   348→  user,
   349→   349→  onUpdate,
   350→   350→}: {
   351→   351→  user: SafeUser
   352→   352→  onUpdate: (u: SafeUser) => void

   356→   356→  const [copiedKey, setCopiedKey] = useState(false)
   357→   357→
   358→   358→  // Edit profile
   359→   359→  const [editOpen, setEditOpen] = useState(false)
   360→   360→  const [displayName, setDisplayName] = useState(user.displayName || '')
   361→   361→  const [savingProfile, setSavingProfile] = useState(false)
   362→   362→
   363→   363→  // Change password
   364→   364→  const [pwOpen, setPwOpen] = useState(false)
   365→   365→  const [currentPw, setCurrentPw] = useState('')
   366→   366→  const [newPw, setNewPw] = useState('')
   367→   367→  const [savingPw, setSavingPw] = useState(false)
   368→   368→
   369→   369→  // Delete account
   370→   370→  const [delOpen, setDelOpen] = useState(false)
   371→   371→  const [delPw, setDelPw] = useState('')
   372→   372→  const [deleting, setDeleting] = useState(false)
   373→   373→
   374→   374→  const { setUser, goHome } = useAppStore()
   375→   375→
   376→   376→  async function regenerate() {
   377→   377→    setRegenerating(true)
   378→   378→    try {
   379→   379→      const rows = await api.updateProfile({ regenerateApiKey: true })
   380→   380→      const u = rows[0]
   381→   381→      if (u) {
   382→   382→        onUpdate(u)
   383→   383→        toast.success('API key regenerated')
   384→   384→      }
   385→   385→    } catch (err) {
   386→   386→      toast.error(err instanceof ApiError ? err.message : 'Failed to regenerate key.')
   387→   387→    } finally {
   388→   388→      setRegenerating(false)
   389→   389→    }
   390→   390→  }
   391→   391→
   392→   392→  function copyKey() {
   393→   393→    if (!user.apiKey) return
   394→   394→    navigator.clipboard.writeText(user.apiKey).then(() => {
   395→   395→      setCopiedKey(true)
   396→   396→      setTimeout(() => setCopiedKey(false), 1500)
   397→   397→    })
   398→   398→  }
   399→   399→
   400→   400→  async function saveProfile() {
   401→   401→    setSavingProfile(true)
   402→   402→    try {
   403→   403→      const rows = await api.updateProfile({ displayName: displayName || undefined })
   404→   404→      const u = rows[0]
   405→   405→      if (u) {
   406→   406→        onUpdate(u)
   407→   407→        toast.success('Profile updated')
   408→   408→        setEditOpen(false)
   409→   409→      }
   410→   410→    } catch (err) {
   411→   411→      toast.error(err instanceof ApiError ? err.message : 'Update failed.')
   412→   412→    } finally {
   413→   413→      setSavingProfile(false)
   414→   414→    }
   415→   415→  }
   416→   416→
   417→   417→  async function savePassword() {
   418→   418→    if (newPw.length < 8) {
   419→   419→      toast.error('New password must be at least 8 characters.')
   420→   420→      return
   421→   421→    }
   422→   422→    setSavingPw(true)
   423→   423→    try {
   424→   424→      const rows = await api.updateProfile({ currentPassword: currentPw, newPassword: newPw })
   425→   425→      const u = rows[0]
   426→   426→      if (u) {
   427→   427→        onUpdate(u)
   428→   428→        toast.success('Password changed')
   429→   429→        setPwOpen(false)
   430→   430→        setCurrentPw('')
   431→   431→        setNewPw('')
   432→   432→      }
   433→   433→    } catch (err) {
   434→   434→      toast.error(err instanceof ApiError ? err.message : 'Password change failed.')
   435→   435→    } finally {
   436→   436→      setSavingPw(false)
   437→   437→    }
   438→   438→  }
   439→   439→
   440→   440→  async function deleteAccount() {
   441→   441→    if (!delPw) {
   442→   442→      toast.error('Enter your current password.')
   443→   443→      return
   444→   444→    }
   445→   445→    setDeleting(true)
   446→   446→    try {
   447→   447→      await api.deleteAccount(delPw)
   448→   448→      toast.success('Account deleted')
   449→   449→      setUser(null)
   450→   450→      goHome()
   451→   451→    } catch (err) {
   452→   452→      toast.error(err instanceof ApiError ? err.message : 'Delete failed.')
   453→   453→    } finally {
   454→   454→      setDeleting(false)
   455→   455→    }
   456→   456→  }
   457→   457→
   458→   458→  const maskedKey = user.apiKey
   459→   459→    ? showKey
   460→   460→      ? user.apiKey
   461→   461→      : user.apiKey.slice(0, 6) + '••••••••••••••••'
   462→   462→    : '—'
   463→   463→
   464→   464→  return (
   465→   465→    <div className="glass-strong gradient-border rounded-2xl p-5 glow-brand-hover">
   466→   466→      {/* Profile head */}
   467→   467→      <div className="flex items-center gap-3">
   468→   468→        <div className="rounded-full bg-gradient-to-br from-brand to-brand-2 p-[2px]">
   469→   469→          <Avatar className="h-12 w-12 border-2 border-background">
   470→   470→            <AvatarFallback className="bg-background text-sm font-semibold text-brand">
   471→   471→              {(user.displayName || user.email).slice(0, 2).toUpperCase()}
   472→   472→            </AvatarFallback>
   473→   473→          </Avatar>
   474→   474→        </div>
   475→   475→        <div className="min-w-0 flex-1">
   476→   476→          <p className="truncate text-base font-semibold">
