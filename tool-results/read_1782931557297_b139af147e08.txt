    61→    61→type PasteWithGate = SafePaste & { passwordRequired?: boolean }
    62→    62→
    63→    63→const MAX_ATTEMPTS = 5
    64→    64→
    65→    65→export function PasteView({ idOrSlug }: { idOrSlug: string }) {
    66→    66→  const goBrowse = useAppStore((s) => s.goBrowse)
    67→    67→  const user = useAppStore((s) => s.user)
    68→    68→
    69→    69→  const [paste, setPaste] = useState<PasteWithGate | null>(null)
    70→    70→  const [loading, setLoading] = useState(true)
    71→    71→  const [notFound, setNotFound] = useState(false)
    72→    72→  const [gone, setGone] = useState(false)
    73→    73→  const [loadError, setLoadError] = useState<string | null>(null)
    74→    74→
    75→    75→  const [passwordInput, setPasswordInput] = useState('')
    76→    76→  const [unlocking, setUnlocking] = useState(false)
    77→    77→  const [unlockError, setUnlockError] = useState<string | null>(null)
    78→    78→  const [attemptsLeft, setAttemptsLeft] = useState(MAX_ATTEMPTS)
    79→    79→
    80→    80→  const [related, setRelated] = useState<SafePaste[]>([])
    81→    81→  const [deleteOpen, setDeleteOpen] = useState(false)
    82→    82→  const [deleting, setDeleting] = useState(false)
    83→    83→
    84→    84→  // Fetch the paste on mount (parent remounts via key on slug change).
    85→    85→  useEffect(() => {
    86→    86→    let alive = true
    87→    87→    setLoading(true)
    88→    88→    setNotFound(false)
    89→    89→    setGone(false)
    90→    90→    setLoadError(null)
    91→    91→    setPaste(null)
    92→    92→    setRelated([])
    93→    93→    setUnlockError(null)
    94→    94→    setAttemptsLeft(MAX_ATTEMPTS)
    95→    95→
    96→    96→    api
    97→    97→      .getPaste(idOrSlug)
    98→    98→      .then((rows) => {
    99→    99→        if (!alive) return
   100→   100→        const p = rows[0]
   101→   101→        if (!p) {
   102→   102→          setNotFound(true)
   103→   103→          return
   104→   104→        }
   105→   105→        setPaste(p)
   106→   106→        if (!p.passwordRequired && p.language) loadRelated(p)
   107→   107→      })
   108→   108→      .catch((e: unknown) => {
   109→   109→        if (!alive) return
   110→   110→        if (e instanceof ApiError) {
   111→   111→          if (e.status === 404) setNotFound(true)
   112→   112→          else if (e.status === 410) setGone(true)
   113→   113→          else setLoadError(e.message)
   114→   114→        } else {
   115→   115→          setLoadError('Failed to load paste')
   116→   116→        }
   117→   117→      })
   118→   118→      .finally(() => {
   119→   119→        if (alive) setLoading(false)
   120→   120→      })
   121→   121→
   122→   122→    function loadRelated(p: SafePaste) {
   123→   123→      api
   124→   124→        .listPastes({ scope: 'language', language: p.language, pageSize: 4 })
   125→   125→        .then((list) => {
   126→   126→          if (alive) setRelated(list.filter((r) => r.id !== p.id).slice(0, 3))
   127→   127→        })
   128→   128→        .catch(() => {
   129→   129→          /* related is optional */
   130→   130→        })
   131→   131→    }
   132→   132→
   133→   133→    return () => {
   134→   134→      alive = false
   135→   135→    }
   136→   136→  }, [idOrSlug])
   137→   137→
   138→   138→  async function handleUnlock(e: React.FormEvent) {
   139→   139→    e.preventDefault()
   140→   140→    if (!paste || unlocking || attemptsLeft <= 0) return
   141→   141→    setUnlocking(true)
   142→   142→    setUnlockError(null)
   143→   143→    try {
   144→   144→      await api.unlockPaste(paste.id, passwordInput)
   145→   145→      markUnlocked(paste.id)
   146→   146→      toast.success('Unlocked')
   147→   147→      setPasswordInput('')
   148→   148→      // Re-fetch — the x-unlocked header is now attached automatically.
   149→   149→      const rows = await api.getPaste(idOrSlug)
   150→   150→      const fresh = rows[0]
   151→   151→      if (fresh) {
   152→   152→        setPaste(fresh)
   153→   153→        if (fresh.language) {
   154→   154→          api
   155→   155→            .listPastes({ scope: 'language', language: fresh.language, pageSize: 4 })
   156→   156→            .then((list) =>
   157→   157→              setRelated(list.filter((r) => r.id !== fresh.id).slice(0, 3)),
   158→   158→            )
   159→   159→            .catch(() => {})
   160→   160→        }
   161→   161→      }
   162→   162→    } catch (e: unknown) {
   163→   163→      const next = attemptsLeft - 1
   164→   164→      setAttemptsLeft(next)
   165→   165→      if (e instanceof ApiError && e.status === 401) {
   166→   166→        setUnlockError(
   167→   167→          next > 0
   168→   168→            ? `Incorrect password. ${next} attempt${next === 1 ? '' : 's'} remaining.`
   169→   169→            : 'No attempts remaining. Please contact the paste owner.',
   170→   170→        )
   171→   171→      } else {
   172→   172→        setUnlockError('Failed to unlock paste. Please try again.')
   173→   173→      }
   174→   174→    } finally {
   175→   175→      setUnlocking(false)
   176→   176→    }
   177→   177→  }
   178→   178→
   179→   179→  async function handleCopy() {
   180→   180→    if (!paste?.textContent) return
   181→   181→    try {
   182→   182→      await navigator.clipboard.writeText(paste.textContent)
   183→   183→      toast.success('Copied to clipboard')
   184→   184→    } catch {
   185→   185→      toast.error('Failed to copy')
   186→   186→    }
   187→   187→  }
   188→   188→
   189→   189→  async function handleShare() {
   190→   190→    if (!paste) return
   191→   191→    try {
   192→   192→      await navigator.clipboard.writeText(paste.slug)
   193→   193→      toast.success(`Share code: ${paste.slug} copied`)
   194→   194→    } catch {
   195→   195→      toast.error('Failed to copy share code')
   196→   196→    }
   197→   197→  }
   198→   198→
   199→   199→  async function handleDownload() {
   200→   200→    if (!paste) return
   201→   201→    const headers: Record<string, string> = {}
   202→   202→    const unlocked = getUnlockedHeader()
   203→   203→    if (unlocked) headers['x-unlocked'] = unlocked
   204→   204→    try {
   205→   205→      const res = await fetch(api.fileDownloadUrl(paste.slug), {
   206→   206→        credentials: 'include',
   207→   207→        headers,
   208→   208→      })
   209→   209→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   210→   210→      const blob = await res.blob()
   211→   211→      const url = URL.createObjectURL(blob)
   212→   212→      const a = document.createElement('a')
   213→   213→      a.href = url
   214→   214→      a.download = paste.originalFileName || paste.slug
   215→   215→      document.body.appendChild(a)
   216→   216→      a.click()
   217→   217→      a.remove()
   218→   218→      URL.revokeObjectURL(url)
   219→   219→      toast.success('Download started')
   220→   220→    } catch {
   221→   221→      toast.error('Download failed')
   222→   222→    }
   223→   223→  }
   224→   224→
   225→   225→  async function handleDelete() {
   226→   226→    if (!paste || deleting) return
   227→   227→    setDeleting(true)
   228→   228→    try {
   229→   229→      await api.deletePaste(paste.id)
   230→   230→      toast.success('Paste deleted')
   231→   231→      goBrowse()
   232→   232→    } catch (e: unknown) {
   233→   233→      toast.error(e instanceof ApiError ? e.message : 'Failed to delete paste')
   234→   234→    } finally {
   235→   235→      setDeleting(false)
   236→   236→      setDeleteOpen(false)
   237→   237→    }
   238→   238→  }
   239→   239→
   240→   240→  // --- State: loading ---
   241→   241→  if (loading) return <PasteSkeleton />
   242→   242→
   243→   243→  // --- State: not found ---
   244→   244→  if (notFound)
   245→   245→    return (
   246→   246→      <StateCard
   247→   247→        icon={SearchX}
   248→   248→        title="Paste not found"
   249→   249→        description="The paste you're looking for doesn't exist, has been deleted, or the link is incorrect."
   250→   250→        onBack={goBrowse}
   251→   251→      />
   252→   252→    )
   253→   253→
   254→   254→  // --- State: gone (expired/burned before any content returned) ---
   255→   255→  if (gone)
   256→   256→    return (
   257→   257→      <StateCard
   258→   258→        icon={Ghost}
   259→   259→        title="This paste is gone"
   260→   260→        description="This paste has expired or was burned after reading and is no longer available."
   261→   261→        onBack={goBrowse}
   262→   262→      />
   263→   263→    )
   264→   264→
   265→   265→  // --- State: other load error ---
   266→   266→  if (loadError)
   267→   267→    return (
   268→   268→      <StateCard
   269→   269→        icon={AlertTriangle}
   270→   270→        title="Couldn't load paste"
   271→   271→        description={loadError}
   272→   272→        onBack={goBrowse}
   273→   273→      />
   274→   274→    )
   275→   275→
   276→   276→  if (!paste) return null
   277→   277→
   278→   278→  // --- State: password gate ---
   279→   279→  if (paste.passwordRequired) {
   280→   280→    return (
   281→   281→      <div className="mx-auto w-full max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
   282→   282→        <Button variant="ghost" size="sm" onClick={goBrowse} className="mb-6 gap-1.5">
   283→   283→          <ArrowLeft className="h-4 w-4" /> Back to browse
   284→   284→        </Button>
   285→   285→        <div className="flex min-h-[60vh] items-center justify-center">
   286→   286→          <Card className="w-full max-w-md">
   287→   287→            <CardHeader className="text-center">
   288→   288→              <div className="mx-auto mb-3 flex h-14 w-14 items-center justify-center rounded-full bg-primary/10">
   289→   289→                <Lock className="h-7 w-7 text-primary" />
   290→   290→              </div>
   291→   291→              <CardTitle className="text-xl">
   292→   292→                {paste.title || `/${paste.slug}`}
   293→   293→              </CardTitle>
   294→   294→              <CardDescription>
   295→   295→                This paste is password-protected. Enter the password to view its contents.
   296→   296→              </CardDescription>
   297→   297→            </CardHeader>
   298→   298→            <CardContent>
   299→   299→              <form onSubmit={handleUnlock} className="space-y-3">
   300→   300→                <Input
   301→   301→                  type="password"
   302→   302→                  value={passwordInput}
   303→   303→                  onChange={(e) => setPasswordInput(e.target.value)}
   304→   304→                  placeholder="Password"
   305→   305→                  autoFocus
   306→   306→                  disabled={unlocking || attemptsLeft <= 0}
   307→   307→                  aria-label="Paste password"
   308→   308→                />
   309→   309→                {unlockError && (
   310→   310→                  <p className="flex items-center gap-1.5 text-sm text-destructive">
   311→   311→                    <AlertTriangle className="h-3.5 w-3.5" /> {unlockError}
   312→   312→                  </p>
   313→   313→                )}
   314→   314→                <div className="flex items-center justify-between text-xs text-muted-foreground">
   315→   315→                  <span className="flex items-center gap-1">
   316→   316→                    <ShieldAlert className="h-3 w-3" />
   317→   317→                    {attemptsLeft} attempt{attemptsLeft === 1 ? '' : 's'} remaining
   318→   318→                  </span>
   319→   319→                  <span className="font-mono">{paste.language}</span>
   320→   320→                </div>
   321→   321→                <Button
   322→   322→                  type="submit"
   323→   323→                  className="w-full"
   324→   324→                  disabled={unlocking || attemptsLeft <= 0 || passwordInput.length === 0}
   325→   325→                >
   326→   326→                  {unlocking ? 'Unlocking…' : 'Unlock paste'}
   327→   327→                </Button>
   328→   328→              </form>
   329→   329→            </CardContent>
   330→   330→          </Card>
   331→   331→        </div>
   332→   332→      </div>
   333→   333→    )
   334→   334→  }
   335→   335→
   336→   336→  // --- State: full view ---
   337→   337→  const isOwner = !!user && !!paste.ownerId && user.id === paste.ownerId
   338→   338→  const visMeta = visibilityMeta(paste.visibility)
   339→   339→  const showBurnedBanner = paste.burnAfterRead && paste.isBurned
   340→   340→  const showBurnWarning = paste.burnAfterRead && !paste.isBurned
   341→   341→  const showExpiredBanner = paste.isExpired && !paste.isBurned
   342→   342→
   343→   343→  return (
   344→   344→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   345→   345→      <Button variant="ghost" size="sm" onClick={goBrowse} className="mb-4 gap-1.5">
   346→   346→        <ArrowLeft className="h-4 w-4" /> Back to browse
   347→   347→      </Button>
   348→   348→
   349→   349→      {/* Status banners */}
   350→   350→      {showBurnedBanner && (
   351→   351→        <AlertBanner tone="amber" icon={Flame} title="This paste was burned after reading">
   352→   352→          It has been permanently deleted from the server. The content shown below is the only
   353→   353→          remaining copy.
   354→   354→        </AlertBanner>
   355→   355→      )}
   356→   356→      {showBurnWarning && (
   357→   357→        <AlertBanner tone="amber" icon={Flame} title="Burn after read — caution">
   358→   358→          Viewing this paste has triggered permanent deletion. It cannot be recovered once you
   359→   359→          leave this page.
   360→   360→        </AlertBanner>
   361→   361→      )}
   362→   362→      {showExpiredBanner && (
   363→   363→        <AlertBanner tone="red" icon={Hourglass} title="This paste has expired">
   364→   364→          It is no longer accessible to new viewers.
   365→   365→        </AlertBanner>
   366→   366→      )}
   367→   367→      {paste.isFlagged && (
   368→   368→        <AlertBanner tone="red" icon={ShieldAlert} title="This paste has been flagged">
   369→   369→          It was flagged by a moderator and may violate community guidelines.
   370→   370→        </AlertBanner>
   371→   371→      )}
   372→   372→
   373→   373→      <div className="grid gap-6 lg:grid-cols-[1fr_20rem]">
   374→   374→        {/* Main column */}
   375→   375→        <div className="min-w-0">
   376→   376→          {/* Title + description + tags */}
   377→   377→          <div className="mb-4">
   378→   378→            <div className="flex flex-wrap items-start gap-2">
   379→   379→              <h1 className="min-w-0 flex-1 text-2xl font-bold tracking-tight sm:text-3xl">
   380→   380→                {paste.title || (paste.contentType === 'file' ? paste.originalFileName : `/${paste.slug}`)}
   381→   381→              </h1>
   382→   382→              <Badge variant="secondary" className="font-mono text-xs">
   383→   383→                {paste.contentType === 'file' ? 'file' : paste.language}
   384→   384→              </Badge>
   385→   385→            </div>
   386→   386→            {paste.description && (
   387→   387→              <p className="mt-2 text-sm text-muted-foreground sm:text-base">
   388→   388→                {paste.description}
   389→   389→              </p>
   390→   390→            )}
   391→   391→            {paste.tags.length > 0 && (
   392→   392→              <div className="mt-3 flex flex-wrap gap-1.5">
   393→   393→                {paste.tags.map((t) => (
   394→   394→                  <Badge key={t} variant="outline" className="text-xs">
   395→   395→                    #{t}
   396→   396→                  </Badge>
   397→   397→                ))}
   398→   398→              </div>
   399→   399→            )}
   400→   400→          </div>
   401→   401→
   402→   402→          {/* Action bar (text pastes only) */}
   403→   403→          {paste.contentType === 'text' && (
   404→   404→            <div className="mb-3 flex flex-wrap items-center gap-2">
   405→   405→              <Button variant="outline" size="sm" onClick={handleCopy} disabled={!paste.textContent}>
   406→   406→                <Copy className="h-4 w-4" /> Copy
   407→   407→              </Button>
   408→   408→              <Button variant="outline" size="sm" asChild>
   409→   409→                <a
   410→   410→                  href={`/api/paste/raw?id=${encodeURIComponent(paste.slug)}`}
   411→   411→                  target="_blank"
   412→   412→                  rel="noreferrer"
   413→   413→                >
   414→   414→                  <Code className="h-4 w-4" /> Raw
   415→   415→                </a>
   416→   416→              </Button>
   417→   417→              <Button variant="outline" size="sm" onClick={handleShare}>
   418→   418→                <Share2 className="h-4 w-4" /> Share
   419→   419→              </Button>
   420→   420→            </div>
   421→   421→          )}
   422→   422→
   423→   423→          {/* Content */}
   424→   424→          {paste.contentType === 'text' ? (
   425→   425→            <div className="overflow-hidden rounded-lg border bg-[#1e1e1e]">
   426→   426→              <CodeEditor
   427→   427→                value={paste.textContent || ''}
   428→   428→                language={paste.language}
   429→   429→                readOnly
   430→   430→                height="600px"
   431→   431→              />
   432→   432→            </div>
   433→   433→          ) : (
   434→   434→            <Card>
   435→   435→              <CardContent className="flex flex-col items-center gap-4 px-6 py-8 text-center sm:flex-row sm:text-left">
   436→   436→                <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-xl bg-primary/10">
   437→   437→                  <FileText className="h-7 w-7 text-primary" />
   438→   438→                </div>
   439→   439→                <div className="min-w-0 flex-1">
   440→   440→                  <p className="truncate font-medium">
   441→   441→                    {paste.originalFileName || `/${paste.slug}`}
   442→   442→                  </p>
   443→   443→                  <p className="mt-0.5 text-sm text-muted-foreground">
   444→   444→                    {formatBytes(paste.fileSizeBytes)}
   445→   445→                    {paste.mimeType ? ` · ${paste.mimeType}` : ''}
   446→   446→                  </p>
   447→   447→                </div>
   448→   448→                <Button onClick={handleDownload} className="gap-1.5">
   449→   449→                  <Download className="h-4 w-4" /> Download
   450→   450→                </Button>
   451→   451→              </CardContent>
   452→   452→            </Card>
   453→   453→          )}
   454→   454→
   455→   455→          {/* Related pastes */}
   456→   456→          {related.length > 0 && (
   457→   457→            <section className="mt-10">
   458→   458→              <h2 className="mb-3 flex items-center gap-2 text-lg font-semibold">
   459→   459→                <Sparkles className="h-4 w-4 text-primary" /> More in {paste.language}
   460→   460→              </h2>
   461→   461→              <div className="grid gap-4 sm:grid-cols-1 lg:grid-cols-3">
   462→   462→                {related.map((p) => (
   463→   463→                  <PasteCard key={p.id} paste={p} />
   464→   464→                ))}
   465→   465→              </div>
   466→   466→            </section>
   467→   467→          )}
   468→   468→        </div>
   469→   469→
   470→   470→        {/* Sidebar */}
   471→   471→        <aside className="lg:sticky lg:top-16 lg:self-start">
   472→   472→          <Card className="rounded-xl">
   473→   473→            <CardHeader className="pb-3">
   474→   474→              <CardTitle className="text-sm">Details</CardTitle>
   475→   475→            </CardHeader>
   476→   476→            <CardContent className="space-y-3">
   477→   477→              {/* Owner */}
   478→   478→              <div className="flex items-center gap-2.5">
   479→   479→                <Avatar className="h-8 w-8">
   480→   480→                  <AvatarFallback className="bg-primary/10 text-xs text-primary">
   481→   481→                    {paste.ownerName
   482→   482→                      ? paste.ownerName.slice(0, 2).toUpperCase()
   483→   483→                      : <UserIcon className="h-3.5 w-3.5" />}
   484→   484→                  </AvatarFallback>
   485→   485→                </Avatar>
   486→   486→                <div className="min-w-0">
   487→   487→                  <p className="truncate text-sm font-medium">
   488→   488→                    {paste.ownerName || 'Guest'}
   489→   489→                  </p>
   490→   490→                  <p className="text-xs text-muted-foreground">Author</p>
   491→   491→                </div>
   492→   492→              </div>
   493→   493→
   494→   494→              <Separator />
   495→   495→
   496→   496→              <MetaRow icon={visMeta.icon} label="Visibility">
   497→   497→                <Badge variant="outline" className="gap-1 text-xs">
   498→   498→                  {visMeta.label}
   499→   499→                </Badge>
   500→   500→              </MetaRow>
   501→   501→
   502→   502→              {paste.contentType === 'text' && (
   503→   503→                <MetaRow icon={Code} label="Language">
   504→   504→                  <span className="font-mono">{paste.language}</span>
   505→   505→                </MetaRow>
   506→   506→              )}
   507→   507→
   508→   508→              <MetaRow icon={Eye} label="Views">{paste.viewCount}</MetaRow>
   509→   509→
   510→   510→              <MetaRow icon={Calendar} label="Created">
   511→   511→                <Tooltip>
   512→   512→                  <TooltipTrigger asChild>
   513→   513→                    <span className="cursor-help">{formatRelativeTime(paste.createdAt)}</span>
   514→   514→                  </TooltipTrigger>
   515→   515→                  <TooltipContent>{formatDate(paste.createdAt)}</TooltipContent>
   516→   516→                </Tooltip>
   517→   517→              </MetaRow>
   518→   518→
   519→   519→              <MetaRow icon={Clock} label="Updated">

   526→   526→              </MetaRow>
   527→   527→
   528→   528→              <MetaRow icon={Hourglass} label="Expires">

   538→   538→                )}
   539→   539→              </MetaRow>
   540→   540→
   541→   541→              <Separator />
   542→   542→
   543→   543→              <div className="flex flex-wrap gap-1.5">

   550→   550→                  <Badge variant="outline" className="gap-1 text-xs">

   555→   555→                  <Badge variant="destructive" className="gap-1 text-xs">
   556→   556→                    <ShieldAlert className="h-3 w-3" /> Flagged
   557→   557→                  </Badge>

   559→   559→                {paste.contentType === 'file' && (

   562→   562→                  </Badge>

   565→   565→
   566→   566→              {/* Owner actions */}
   567→   567→              {isOwner && (
   568→   568→                <>
   569→   569→                  <Separator />
   570→   570→                  <div className="flex flex-col gap-2 pt-1">
   571→   571→                    <Button
   572→   572→                      variant="outline"

   584→   584→                        size="sm"
   585→   585→                        className="w-full gap-1.5"
   586→   586→                        onClick={() => setDeleteOpen(true)}
   587→   587→                      >
   588→   588→                        <Trash2 className="h-4 w-4" /> Delete
   589→   589→                      </Button>
   590→   590→                      <AlertDialogContent>
   591→   591→                        <AlertDialogHeader>
   592→   592→                          <AlertDialogTitle>Delete this paste?</AlertDialogTitle>

   597→   597→                        </AlertDialogHeader>

   613→   613→            </CardContent>
   614→   614→            <CardFooter className="pt-0">
   615→   615→              <p className="w-full text-center font-mono text-[10px] text-muted-foreground">
   616→   616→                /{paste.slug}
   617→   617→              </p>
   618→   618→            </CardFooter>
   619→   619→          </Card>
   620→   620→        </aside>
   621→   621→      </div>
   622→   622→    </div>
   623→   623→  )
