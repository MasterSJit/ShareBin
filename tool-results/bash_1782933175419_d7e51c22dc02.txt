   671→                  <TooltipContent>Login required</TooltipContent>
   672→                </Tooltip>
   673→              )
   674→            }
   675→            return <div key={opt.value}>{card}</div>
   676→          })}
   677→        </RadioGroup>
   678→      </div>
   679→
   680→      {/* Password protection */}
   681→      <div className="space-y-2">
   682→        <div className="flex items-center justify-between">
   683→          <Label
   684→            htmlFor="pw-switch"
   685→            className="flex items-center gap-1.5 text-xs font-medium"
   686→          >
   687→            <Lock className="h-3.5 w-3.5 text-muted-foreground" /> Password
   688→          </Label>
   689→          <Switch
   690→            id="pw-switch"
   691→            checked={passwordEnabled}
   692→            onCheckedChange={setPasswordEnabled}
   693→            disabled={burnAfterRead}
   694→            className="data-[state=checked]:bg-brand"
   695→          />
   696→        </div>
   697→        {passwordEnabled && (
   698→          <div className="relative">
   699→            <Lock className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   700→            <Input
   701→              type="password"
   702→              value={password}
   703→              onChange={(e) => setPassword(e.target.value)}
   704→              placeholder="Password"
   705→              className="pl-8 text-sm"
   706→              autoComplete="new-password"
   707→            />
   708→          </div>
   709→        )}
   710→        {burnAfterRead && (
   711→          <p className="text-[11px] text-muted-foreground">
   712→            Cannot combine with burn-after-read.
   713→          </p>
   714→        )}
   715→      </div>
   716→
   717→      {/* Burn after read */}
   718→      <div className="space-y-2">
   719→        <div className="flex items-center justify-between">
   720→          <Label
   721→            htmlFor="burn-switch"
   722→            className="flex items-center gap-1.5 text-xs font-medium"
   723→          >
   724→            <Flame className="h-3.5 w-3.5 text-muted-foreground" /> Burn after read
   725→          </Label>
   726→          <Switch
   727→            id="burn-switch"
   728→            checked={burnAfterRead}
   729→            onCheckedChange={setBurnAfterRead}
   730→            disabled={passwordEnabled}
   731→            className="data-[state=checked]:bg-brand-2"
   732→          />
   733→        </div>
   734→        {passwordEnabled && (
   735→          <p className="text-[11px] text-muted-foreground">
   736→            Cannot combine with password.
   737→          </p>
   738→        )}
   739→      </div>
   740→
   741→      {/* Expiry */}
   742→      <div className="space-y-2">
   743→        <Label
   744→          htmlFor="expiry-select"
   745→          className="flex items-center gap-1.5 text-xs font-medium"
   746→        >
   747→          <Clock className="h-3.5 w-3.5 text-muted-foreground" /> Expiry
   748→        </Label>
   749→        <Select value={expiryPreset} onValueChange={setExpiryPreset}>
   750→          <SelectTrigger id="expiry-select" className="w-full bg-muted/30">
   751→            <SelectValue />
   752→          </SelectTrigger>
   753→          <SelectContent>
   754→            {EXPIRY_OPTIONS.map((opt) => (
   755→              <SelectItem key={opt.value} value={opt.value}>
   756→                {opt.label}
   757→              </SelectItem>
   758→            ))}
   759→          </SelectContent>
   760→        </Select>
   761→        {expiryPreset === 'custom' && (
   762→          <Input
   763→            type="datetime-local"
   764→            value={expiryCustom}
   765→            onChange={(e) => setExpiryCustom(e.target.value)}
   766→            className="text-sm"
   767→            aria-label="Custom expiry date and time"
   768→          />
   769→        )}
   770→      </div>
   771→
   772→      {/* Submit */}
   773→      <Button
   774→        type="button"
   775→        onClick={onSubmit}
   776→        disabled={submitDisabled || submitting}
   777→        className="btn-shine relative w-full gap-1.5 overflow-hidden bg-gradient-to-r from-brand to-brand-2 text-brand-foreground shadow-lg shadow-brand/30 hover:shadow-brand/50"
   778→        size="lg"
   779→      >
   780→        {submitting ? (
   781→          <Loader2 className="h-4 w-4 animate-spin" />
   782→        ) : (
   783→          submitIcon
   784→        )}
   785→        {submitLabel}
   786→      </Button>
   787→
   788→      {!user && (
   789→        <p className="text-center text-[11px] text-muted-foreground">
   790→          Guest paste —{' '}
   791→          <button
   792→            type="button"
   793→            onClick={onLogin}
   794→            className="font-medium text-brand hover:underline"
   795→          >
   796→            log in
   797→          </button>{' '}
   798→          for private pastes
   799→        </p>
   800→      )}
   801→    </div>
   802→  )
   803→}
