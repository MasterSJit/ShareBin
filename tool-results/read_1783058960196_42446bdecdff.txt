   321→/** Reverse-engineer the expiry preset from a paste's expiresAt vs createdAt. */
   322→function deriveExpiryFromPaste(p: SafePaste): {
   323→  preset: string
   324→  custom: string
   325→} {
   326→  if (!p.expiresAt) return { preset: 'never', custom: '' }
   327→  const created = new Date(p.createdAt).getTime()
   328→  const expires = new Date(p.expiresAt).getTime()
   329→  const diffMs = expires - created
   330→  for (const [value, ms] of Object.entries(PRESET_MS)) {
   331→    if (Math.abs(diffMs - ms) < 60 * 1000) {
   332→      return { preset: value, custom: '' }
   333→    }
   334→  }
   335→  // Custom — fill datetime-local in local time.
   336→  const dt = new Date(p.expiresAt)
   337→  const pad = (n: number) => String(n).padStart(2, '0')
   338→  const custom = `${dt.getFullYear()}-${pad(dt.getMonth() + 1)}-${pad(
   339→    dt.getDate(),
   340→  )}T${pad(dt.getHours())}:${pad(dt.getMinutes())}`
   341→  return { preset: 'custom', custom }
   342→}
   343→
   344→/** Format a Date into a value usable by `<input type="datetime-local">`. */
   345→function toLocalDatetime(d: Date): string {
   346→  const pad = (n: number) => String(n).padStart(2, '0')
   347→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(
   348→    d.getDate(),
   349→  )}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   350→}

   881→function UsersTab() {
   882→  const me = useAppStore((s) => s.user)!
   883→  const [rows, setRows] = useState<UserRow[]>([])
   884→  const [loading, setLoading] = useState(true)
   885→  const [q, setQ] = useState('')
   886→  const [status, setStatus] = useState<string>('all')
   887→  const [confirm, setConfirm] = useState<{
   888→    user: UserRow
   889→    action: 'delete' | 'ban' | 'suspend'
   890→  } | null>(null)
   891→  const [busy, setBusy] = useState(false)
   892→
   893→  async function fetchUsers() {
   894→    setLoading(true)
   895→    try {
   896→      const r = await api.admin.listUsers(
   897→        q || undefined,
   898→        status === 'all' ? undefined : status,
   899→      )
   900→      setRows(r)
   901→    } catch (err) {
   902→      toast.error(err instanceof ApiError ? err.message : 'Failed to load users.')
   903→    } finally {
   904→      setLoading(false)
   905→    }
   906→  }
   907→
   908→  useEffect(() => {
   909→    fetchUsers()
   910→  }, [status])
   911→
   912→  async function patchUser(
   913→    u: UserRow,
   914→    patch: { status?: string; role?: string; verify?: boolean },
   915→  ) {
   916→    if (u.id === me.id && (patch.status === 'banned' || patch.status === 'suspended')) {
   917→      toast.error('You cannot suspend or ban yourself.')
   918→      return
   919→    }
   920→    try {
   921→      const res = await api.admin.updateUser(u.id, patch)
   922→      const updated = res[0] as UserRow | undefined
   923→      if (updated) {
   924→        setRows((cur) => cur.map((x) => (x.id === u.id ? { ...x, ...updated } : x)))
   925→        toast.success('User updated')
   926→      }
   927→    } catch (err) {
   928→      toast.error(err instanceof ApiError ? err.message : 'Update failed.')
   929→    }
   930→  }
   931→
   932→  async function doConfirm() {
   933→    if (!confirm) return
   934→    setBusy(true)
   935→    try {
   936→      if (confirm.action === 'delete') {
   937→        if (confirm.user.id === me.id) {
   938→          toast.error('You cannot delete yourself.')
   939→          return
   940→        }
   941→        await api.admin.deleteUser(confirm.user.id)
   942→        setRows((cur) => cur.filter((x) => x.id !== confirm.user.id))
   943→        toast.success('User deleted')
   944→      } else if (confirm.action === 'ban') {
   945→        if (confirm.user.id === me.id) {
   946→          toast.error('You cannot ban yourself.')
   947→          return
   948→        }
   949→        const res = await api.admin.updateUser(confirm.user.id, { status: 'banned' })
   950→        const updated = res[0] as UserRow | undefined
   951→        if (updated)
   952→          setRows((cur) =>
   953→            cur.map((x) => (x.id === confirm.user.id ? { ...x, ...updated } : x)),
   954→          )
   955→        toast.success('User banned')
   956→      } else if (confirm.action === 'suspend') {
   957→        if (confirm.user.id === me.id) {
   958→          toast.error('You cannot suspend yourself.')
   959→          return
   960→        }
   961→        const res = await api.admin.updateUser(confirm.user.id, { status: 'suspended' })
   962→        const updated = res[0] as UserRow | undefined
   963→        if (updated)
   964→          setRows((cur) =>
   965→            cur.map((x) => (x.id === confirm.user.id ? { ...x, ...updated } : x)),
   966→          )
   967→        toast.success('User suspended')
   968→      }
   969→    } catch (err) {
   970→      toast.error(err instanceof ApiError ? err.message : 'Action failed.')
   971→    } finally {
   972→      setBusy(false)
   973→      setConfirm(null)
   974→    }
   975→  }
   976→
   977→  return (
   978→    <div className="space-y-4">
   979→      {/* Filter bar */}
   980→      <Reveal variant={fadeUp}>
   981→        <div className="surface flex flex-wrap items-center gap-2 rounded-xl p-3">
   982→          <div className="relative min-w-[200px] flex-1">
   983→            <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   984→            <Input
   985→              placeholder="Search by email or name…"
   986→              value={q}
   987→              onChange={(e) => setQ(e.target.value)}
   988→              onKeyDown={(e) => e.key === 'Enter' && fetchUsers()}
   989→              className={`${INPUT_CLASS} pl-9`}
   990→            />
   991→          </div>
   992→          <Select value={status} onValueChange={setStatus}>
   993→            <SelectTrigger className={`${INPUT_CLASS} w-[140px]`}>
   994→              <SelectValue placeholder="Status" />
   995→            </SelectTrigger>
   996→            <SelectContent>
   997→              <SelectItem value="all">All</SelectItem>
   998→              <SelectItem value="active">Active</SelectItem>
   999→              <SelectItem value="pending_verification">Pending</SelectItem>
  1000→              <SelectItem value="suspended">Suspended</SelectItem>
  1001→              <SelectItem value="banned">Banned</SelectItem>
  1002→            </SelectContent>
  1003→          </Select>
  1004→          <Button
  1005→            onClick={fetchUsers}
  1006→            className="bg-brand text-brand-foreground hover:bg-brand/90"
  1007→            size="sm"
  1008→          >
  1009→            <Search className="h-3.5 w-3.5" /> Search
  1010→          </Button>
  1011→        </div>
  1012→      </Reveal>
  1013→
  1014→      {/* Table */}

  1018→            <Table>

  1020→                <TableRow className="border-border hover:bg-transparent">
  1021→                  <TableHead className="font-mono px-4 text-[10px] uppercase tracking-wider text-muted-foreground">
