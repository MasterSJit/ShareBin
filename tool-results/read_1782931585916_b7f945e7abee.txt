   358→const STATUS_OPTIONS = [
   359→  { value: 'all', label: 'All statuses' },
   360→  { value: 'active', label: 'Active' },
   361→  { value: 'suspended', label: 'Suspended' },

   366→function UsersTab() {
   367→  const me = useAppStore((s) => s.user)
   368→  const [q, setQ] = useState('')
   369→  const [status, setStatus] = useState('all')
   370→  const [rows, setRows] = useState<UserRow[]>([])
   371→  const [loading, setLoading] = useState(true)
   372→  const [deleteTarget, setDeleteTarget] = useState<UserRow | null>(null)
   373→  const [deleting, setDeleting] = useState(false)
   374→
   375→  async function fetchUsers() {
   376→    setLoading(true)
   377→    try {
   378→      const data = await api.admin.listUsers(q || undefined, status === 'all' ? undefined : status)
   379→      setRows(data)
   380→    } catch (err) {
   381→      toast.error(err instanceof ApiError ? err.message : 'Failed to load users.')
   382→    } finally {
   383→      setLoading(false)
   384→    }
   385→  }
   386→
   387→  useEffect(() => {
   388→    const t = setTimeout(fetchUsers, 250)
   389→    return () => clearTimeout(t)
   390→  }, [q, status])
   391→
   392→  async function update(id: string, patch: Parameters<typeof api.admin.updateUser>[1]) {
   393→    if (me && id === me.id) {
   394→      const risky =
   395→        patch.status === 'banned' ||
   396→        patch.status === 'suspended' ||
   397→        patch.role === 'user'
   398→      if (risky) {
   399→        toast.error("You can't modify your own role or status.")
   400→        return
   401→      }
   402→    }
   403→    try {
   404→      await api.admin.updateUser(id, patch)
   405→      toast.success('User updated')
   406→      fetchUsers()
   407→    } catch (err) {
   408→      toast.error(err instanceof ApiError ? err.message : 'Update failed.')
   409→    }
   410→  }
   411→
   412→  async function doDelete() {
   413→    if (!deleteTarget) return
   414→    if (me && deleteTarget.id === me.id) {
   415→      toast.error("You can't delete your own account from here.")
   416→      setDeleteTarget(null)
   417→      return
   418→    }
   419→    setDeleting(true)
   420→    try {
   421→      await api.admin.deleteUser(deleteTarget.id)
   422→      toast.success('User deleted')
   423→      fetchUsers()
   424→    } catch (err) {
   425→      toast.error(err instanceof ApiError ? err.message : 'Delete failed.')
   426→    } finally {
   427→      setDeleting(false)
   428→      setDeleteTarget(null)
   429→    }
   430→  }
   431→
   432→  return (
   433→    <Card>
   434→      <CardHeader>
   435→        <CardTitle className="text-base">Users</CardTitle>
   436→        <CardDescription>{rows.length} user(s) match the current filter.</CardDescription>
   437→      </CardHeader>
   438→      <CardContent className="space-y-4">
   439→        <div className="flex flex-wrap gap-2">
   440→          <div className="relative flex-1 min-w-[200px]">
   441→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   442→            <Input
   443→              placeholder="Search by name or email…"
   444→              value={q}
   445→              onChange={(e) => setQ(e.target.value)}
   446→              className="pl-8"
   447→            />
   448→          </div>
   449→          <Select value={status} onValueChange={setStatus}>
   450→            <SelectTrigger className="w-[200px]">
   451→              <SelectValue />
   452→            </SelectTrigger>
   453→            <SelectContent>
   454→              {STATUS_OPTIONS.map((o) => (
   455→                <SelectItem key={o.value} value={o.value}>
   456→                  {o.label}
   457→                </SelectItem>
   458→              ))}
   459→            </SelectContent>
   460→          </Select>
   461→        </div>
   462→
   463→        <div className={`max-h-[60vh] overflow-y-auto rounded-md border ${SCROLLBAR_CLASS}`}>
   464→          <Table>
   465→            <TableHeader className="sticky top-0 bg-card z-10">
   466→              <TableRow>
   467→                <TableHead>User</TableHead>
   468→                <TableHead>Role</TableHead>
   469→                <TableHead>Status</TableHead>
   470→                <TableHead className="text-right">Pastes</TableHead>
   471→                <TableHead>Created</TableHead>
   472→                <TableHead className="w-10"></TableHead>
   473→              </TableRow>
   474→            </TableHeader>
   475→            <TableBody>
   476→              {loading ? (
   477→                [0, 1, 2, 3].map((i) => (
   478→                  <TableRow key={i}>
   479→                    <TableCell colSpan={6}>
   480→                      <Skeleton className="h-8 w-full" />
   481→                    </TableCell>
   482→                  </TableRow>
   483→                ))
   484→              ) : rows.length === 0 ? (
   485→                <TableRow>
   486→                  <TableCell colSpan={6} className="text-center text-muted-foreground py-8">
   487→                    No users found.
   488→                  </TableCell>
   489→                </TableRow>
   490→              ) : (
   491→                rows.map((u) => (
   492→                  <TableRow key={u.id}>
   493→                    <TableCell>
   494→                      <div className="flex items-center gap-2">
   495→                        <Avatar className="h-7 w-7">
   496→                          <AvatarFallback className="bg-primary/10 text-primary text-xs">
   497→                            {(u.displayName || u.email).slice(0, 2).toUpperCase()}
   498→                          </AvatarFallback>
   499→                        </Avatar>
   500→                        <div className="min-w-0">
   501→                          <p className="text-sm font-medium truncate">
   502→                            {u.displayName || 'Unnamed'}
   503→                          </p>
   504→                          <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   505→                        </div>
   506→                      </div>
   507→                    </TableCell>
   508→                    <TableCell>
   509→                      {u.role === 'admin' ? (
   510→                        <Badge variant="default" className="gap-1">
   511→                          <Crown className="h-3 w-3" /> admin
   512→                        </Badge>
   513→                      ) : (
   514→                        <Badge variant="outline">user</Badge>
   515→                      )}
   516→                    </TableCell>
   517→                    <TableCell>
   518→                      <StatusBadge status={u.status} />
   519→                    </TableCell>
   520→                    <TableCell className="text-right font-mono text-sm">
   521→                      {u.pasteCount ?? 0}
   522→                    </TableCell>
   523→                    <TableCell className="text-xs text-muted-foreground">
   524→                      {formatRelativeTime(u.createdAt)}
   525→                    </TableCell>
   526→                    <TableCell>
   527→                      <DropdownMenu>
   528→                        <DropdownMenuTrigger asChild>
   529→                          <Button size="icon" variant="ghost" className="h-7 w-7">
   530→                            <MoreHorizontal className="h-4 w-4" />
   531→                          </Button>
   532→                        </DropdownMenuTrigger>
   533→                        <DropdownMenuContent align="end" className="w-48">
   534→                          <DropdownMenuLabel>Actions</DropdownMenuLabel>
   535→                          {u.status !== 'active' && (
   536→                            <DropdownMenuItem onClick={() => update(u.id, { status: 'active', verify: true })}>
   537→                              <CheckCircle2 className="mr-2 h-3.5 w-3.5" /> Verify / activate
   538→                            </DropdownMenuItem>
   539→                          )}
   540→                          {u.status !== 'suspended' && (
   541→                            <DropdownMenuItem onClick={() => update(u.id, { status: 'suspended' })}>
   542→                              <Pause className="mr-2 h-3.5 w-3.5" /> Suspend
   543→                            </DropdownMenuItem>
   544→                          )}
   545→                          {u.status !== 'banned' && (
   546→                            <DropdownMenuItem onClick={() => update(u.id, { status: 'banned' })}>
   547→                              <Ban className="mr-2 h-3.5 w-3.5" /> Ban
   548→                            </DropdownMenuItem>
   549→                          )}
   550→                          <DropdownMenuSeparator />
   551→                          {u.role !== 'admin' ? (
   552→                            <DropdownMenuItem onClick={() => update(u.id, { role: 'admin' })}>
   553→                              <ArrowUpCircle className="mr-2 h-3.5 w-3.5" /> Make admin
   554→                            </DropdownMenuItem>
   555→                          ) : (
   556→                            <DropdownMenuItem onClick={() => update(u.id, { role: 'user' })}>
   557→                              <ArrowDownCircle className="mr-2 h-3.5 w-3.5" /> Demote to user
   558→                            </DropdownMenuItem>
   559→                          )}
   560→                          <DropdownMenuSeparator />
   561→                          <DropdownMenuItem
   562→                            className="text-destructive focus:text-destructive"
