    76→    76→  const user = useAppStore((s) => s.user)

    78→    78→  const [paste, setPaste] = useState<PasteWithGate | null>(null)

    82→    82→  const [loadError, setLoadError] = useState<string | null>(null)
    83→    83→
    84→    84→  const [passwordInput, setPasswordInput] = useState('')
    85→    85→  const [unlocking, setUnlocking] = useState(false)
    86→    86→  const [unlockError, setUnlockError] = useState<string | null>(null)
    87→    87→  const [attemptsLeft, setAttemptsLeft] = useState(MAX_ATTEMPTS)
    88→    88→  const [shakeKey, setShakeKey] = useState(0)
    89→    89→
    90→    90→  const [related, setRelated] = useState<SafePaste[]>([])
    91→    91→  const [deleteOpen, setDeleteOpen] = useState(false)
    92→    92→  const [deleting, setDeleting] = useState(false)
    93→    93→
    94→    94→  // Fetch the paste on mount (parent remounts via key on slug change).
    95→    95→  useEffect(() => {
    96→    96→    let alive = true
    97→    97→    setLoading(true)
    98→    98→    setNotFound(false)
    99→    99→    setGone(false)
   100→   100→    setLoadError(null)
   101→   101→    setPaste(null)
   102→   102→    setRelated([])
   103→   103→    setUnlockError(null)
   104→   104→    setAttemptsLeft(MAX_ATTEMPTS)
   105→   105→
   106→   106→    api
   107→   107→      .getPaste(idOrSlug)
   108→   108→      .then((rows) => {
   109→   109→        if (!alive) return
   110→   110→        const p = rows[0]
   111→   111→        if (!p) {
   112→   112→          setNotFound(true)
   113→   113→          return
   114→   114→        }
   115→   115→        setPaste(p)
   116→   116→        if (!p.passwordRequired && p.language) loadRelated(p)
   117→   117→      })
   118→   118→      .catch((e: unknown) => {
   119→   119→        if (!alive) return
   120→   120→        if (e instanceof ApiError) {
   121→   121→          if (e.status === 404) setNotFound(true)
   122→   122→          else if (e.status === 410) setGone(true)
   123→   123→          else setLoadError(e.message)
   124→   124→        } else {
   125→   125→          setLoadError('Failed to load paste')
   126→   126→        }
   127→   127→      })
   128→   128→      .finally(() => {
   129→   129→        if (alive) setLoading(false)
   130→   130→      })
   131→   131→
   132→   132→    function loadRelated(p: SafePaste) {
   133→   133→      api
   134→   134→        .listPastes({ scope: 'language', language: p.language, pageSize: 4 })
   135→   135→        .then((list) => {
   136→   136→          if (alive) setRelated(list.filter((r) => r.id !== p.id).slice(0, 3))
   137→   137→        })
   138→   138→        .catch(() => {
   139→   139→          /* related is optional */
   140→   140→        })
   141→   141→    }
   142→   142→
   143→   143→    return () => {
   144→   144→      alive = false
   145→   145→    }
   146→   146→  }, [idOrSlug])
   147→   147→
   148→   148→  async function handleUnlock(e: React.FormEvent) {
   149→   149→    e.preventDefault()

   151→   151→    setUnlocking(true)
   152→   152→    setUnlockError(null)
   153→   153→    try {
   154→   154→      await api.unlockPaste(paste.id, passwordInput)
   155→   155→      markUnlocked(paste.id)
   156→   156→      toast.success('Unlocked 🔓')
   157→   157→      setPasswordInput('')
   158→   158→      // Re-fetch — the x-unlocked header is now attached automatically.
   159→   159→      const rows = await api.getPaste(idOrSlug)
   160→   160→      const fresh = rows[0]
   161→   161→      if (fresh) {
   162→   162→        setPaste(fresh)
   163→   163→        if (fresh.language) {
   164→   164→          api
   165→   165→            .listPastes({ scope: 'language', language: fresh.language, pageSize: 4 })
   166→   166→            .then((list) =>
   167→   167→              setRelated(list.filter((r) => r.id !== fresh.id).slice(0, 3)),
   168→   168→            )
   169→   169→            .catch(() => {})
   170→   170→        }
   171→   171→      }
   172→   172→    } catch (e: unknown) {
   173→   173→      const next = attemptsLeft - 1
   174→   174→      setAttemptsLeft(next)
   175→   175→      setShakeKey((k) => k + 1)
   176→   176→      if (e instanceof ApiError && e.status === 401) {
   177→   177→        setUnlockError(
   178→   178→          next > 0
   179→   179→            ? `Incorrect password. ${next} attempt${next === 1 ? '' : 's'} remaining.`
   180→   180→            : 'No attempts remaining. Please contact the paste owner.',
   181→   181→        )
   182→   182→      } else {
   183→   183→        setUnlockError('Failed to unlock paste. Please try again.')
   184→   184→      }
   185→   185→    } finally {
   186→   186→      setUnlocking(false)
   187→   187→    }
   188→   188→  }
   189→   189→
   190→   190→  async function handleCopy() {
   191→   191→    if (!paste?.textContent) return
   192→   192→    try {
   193→   193→      await navigator.clipboard.writeText(paste.textContent)
   194→   194→      toast.success('Copied to clipboard')
   195→   195→    } catch {
   196→   196→      toast.error('Failed to copy')
   197→   197→    }
   198→   198→  }
   199→   199→
   200→   200→  async function handleShare() {
   201→   201→    if (!paste) return
   202→   202→    try {
   203→   203→      await navigator.clipboard.writeText(paste.slug)
   204→   204→      toast.success(`Share code: ${paste.slug} copied`)
   205→   205→    } catch {
   206→   206→      toast.error('Failed to copy share code')
   207→   207→    }
   208→   208→  }
   209→   209→
   210→   210→  async function handleDownload() {
   211→   211→    if (!paste) return
   212→   212→    const headers: Record<string, string> = {}
   213→   213→    const unlocked = getUnlockedHeader()
   214→   214→    if (unlocked) headers['x-unlocked'] = unlocked
   215→   215→    try {
   216→   216→      const res = await fetch(api.fileDownloadUrl(paste.slug), {
   217→   217→        credentials: 'include',
   218→   218→        headers,
   219→   219→      })
   220→   220→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   221→   221→      const blob = await res.blob()
   222→   222→      const url = URL.createObjectURL(blob)
   223→   223→      const a = document.createElement('a')
   224→   224→      a.href = url
   225→   225→      a.download = paste.originalFileName || paste.slug
   226→   226→      document.body.appendChild(a)
   227→   227→      a.click()
   228→   228→      a.remove()
   229→   229→      URL.revokeObjectURL(url)
   230→   230→      toast.success('Download started')
   231→   231→    } catch {
   232→   232→      toast.error('Download failed')
   233→   233→    }
   234→   234→  }
   235→   235→
   236→   236→  async function handleDelete() {
   237→   237→    if (!paste || deleting) return

   240→   240→      await api.deletePaste(paste.id)

   242→   242→      goBrowse()
   243→   243→    } catch (e: unknown) {
   244→   244→      toast.error(e instanceof ApiError ? e.message : 'Failed to delete paste')
   245→   245→    } finally {
   246→   246→      setDeleting(false)
   247→   247→      setDeleteOpen(false)
   248→   248→    }
   249→   249→  }
   250→   250→
   251→   251→  // --- State: loading ---
   252→   252→  if (loading) return <PasteSkeleton />
   253→   253→
   254→   254→  // --- State: not found ---

   256→   256→    return (
   257→   257→      <StateCard

   259→   259→        title="Paste not found"
   260→   260→        description="The paste you're looking for doesn't exist, has been deleted, or the link is incorrect."
   261→   261→        onBack={goBrowse}
   262→   262→      />
   263→   263→    )
   264→   264→
   265→   265→  // --- State: gone (expired/burned before any content returned) ---
   266→   266→  if (gone)
   267→   267→    return (
   268→   268→      <StateCard
   269→   269→        icon={Flame}
   270→   270→        title="This paste has expired or been burned."
   271→   271→        description="It is no longer available. Burned pastes are permanently deleted after their first read."
   272→   272→        onBack={goBrowse}
   273→   273→      />
   274→   274→    )
   275→   275→
   276→   276→  // --- State: other load error ---
   277→   277→  if (loadError)
   278→   278→    return (
   279→   279→      <StateCard
   280→   280→        icon={AlertTriangle}
   281→   281→        title="Couldn't load paste"
   282→   282→        description={loadError}
   283→   283→        onBack={goBrowse}
   284→   284→      />
   285→   285→    )
   286→   286→
   287→   287→  if (!paste) return null
   288→   288→
   289→   289→  // --- State: password gate ---

   291→   291→    const lowAttempts = attemptsLeft <= 2
   292→   292→    return (

   295→   295→          variant="ghost"
   296→   296→          size="sm"

   298→   298→          className="mb-6 gap-1.5 text-muted-foreground hover:text-foreground"

   304→   304→            key={shakeKey}
   305→   305→            initial={{ opacity: 0, y: 16, scale: 0.96 }}
   306→   306→            animate={{
   307→   307→              opacity: 1,

   311→   311→            }}
   312→   312→            transition={{
   313→   313→              opacity: { duration: 0.4, ease: [0.2, 0.8, 0.2, 1] },
   314→   314→              y: { duration: 0.4, ease: [0.2, 0.8, 0.2, 1] },
   315→   315→              scale: { duration: 0.4, ease: [0.2, 0.8, 0.2, 1] },
   316→   316→              x: { duration: 0.45, ease: 'easeInOut' },

   318→   318→            className="glass-strong glow-brand gradient-border w-full max-w-md rounded-2xl"
   319→   319→          >
   320→   320→            <CardHeader className="items-center text-center">
   321→   321→              <div className="animate-pulse-glow relative mx-auto mb-3 flex h-16 w-16 items-center justify-center rounded-full bg-gradient-to-br from-brand to-brand-2 shadow-lg shadow-brand/30">
   322→   322→                <Lock className="h-7 w-7 text-brand-foreground" />

   336→   336→                  <Input
   337→   337→                    type="password"
   338→   338→                    value={passwordInput}
   339→   339→                    onChange={(e) => setPasswordInput(e.target.value)}

   348→   348→                {unlockError && (
   349→   349→                  <motion.p

   357→   357→
   358→   358→                <div className="flex items-center justify-between text-xs">

   368→   368→                </div>
   369→   369→
   370→   370→                <Button
   371→   371→                  type="submit"
   372→   372→                  disabled={unlocking || attemptsLeft <= 0 || passwordInput.length === 0}

   375→   375→                  {unlocking ? (

   377→   377→                      <Loader2 className="h-4 w-4 animate-spin" /> Unlocking…
   378→   378→                    </>
   379→   379→                  ) : (
   380→   380→                    <>
   381→   381→                      <Lock className="h-4 w-4" /> Unlock paste
   382→   382→                    </>
   383→   383→                  )}
   384→   384→                </Button>
   385→   385→              </form>
   386→   386→            </CardContent>
   387→   387→          </motion.div>
   388→   388→        </div>
   389→   389→      </div>
   390→   390→    )
   391→   391→  }
   392→   392→
   393→   393→  // --- State: full view ---
   394→   394→  const isOwner = !!user && !!paste.ownerId && user.id === paste.ownerId
   395→   395→  const visMeta = visibilityMeta(paste.visibility)
   396→   396→  const showBurnBanner = paste.burnAfterRead
   397→   397→
   398→   398→  return (
   399→   399→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   400→   400→      <Reveal className="mb-4">
   401→   401→        <Button
   402→   402→          variant="ghost"
   403→   403→          size="sm"
   404→   404→          onClick={goBrowse}
   405→   405→          className="gap-1.5 text-muted-foreground hover:text-foreground"
   406→   406→        >
   407→   407→          <ArrowLeft className="h-4 w-4" /> Back to browse
   408→   408→        </Button>
   409→   409→      </Reveal>
   410→   410→
   411→   411→      {/* Burn-after-read warning banner */}
   412→   412→      {showBurnBanner && (
   413→   413→        <Reveal className="mb-4">
   414→   414→          <div
   415→   415→            className="glass animate-pulse-glow flex items-start gap-3 rounded-xl border border-brand-2/40 p-4"
   416→   416→            role="alert"
   417→   417→          >
   418→   418→            <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-amber-500/20 to-brand-2/20">
   419→   419→              <Flame className="h-5 w-5 text-amber-500" />
   420→   420→            </div>
   421→   421→            <div className="min-w-0">
   422→   422→              <p className="font-semibold text-foreground">
   423→   423→                This paste was burned after reading — it is now permanently deleted.
   424→   424→              </p>
   425→   425→              <p className="mt-0.5 text-sm text-muted-foreground">
   426→   426→                The content shown below is the only remaining copy. It cannot be recovered once you leave this page.
   427→   427→              </p>
   428→   428→            </div>
   429→   429→          </div>
   430→   430→        </Reveal>
   431→   431→      )}
   432→   432→
   433→   433→      {/* Expired / flagged banners */}
   434→   434→      {paste.isExpired && !paste.isBurned && (
   435→   435→        <Reveal className="mb-4">
   436→   436→          <div

   438→   438→            role="alert"
   439→   439→          >
   440→   440→            <Hourglass className="mt-0.5 h-5 w-5 shrink-0" />
   441→   441→            <div className="min-w-0">
   442→   442→              <p className="font-semibold">This paste has expired</p>
   443→   443→              <p className="mt-0.5 text-sm opacity-90">
   444→   444→                It is no longer accessible to new viewers.
   445→   445→              </p>
   446→   446→            </div>
   447→   447→          </div>
   448→   448→        </Reveal>
   449→   449→      )}
   450→   450→      {paste.isFlagged && (
   451→   451→        <Reveal className="mb-4">
   452→   452→          <div
   453→   453→            className="flex items-start gap-3 rounded-xl border border-destructive/40 bg-destructive/10 p-4 text-destructive"
   454→   454→            role="alert"
   455→   455→          >
   456→   456→            <Flag className="mt-0.5 h-5 w-5 shrink-0" />
   457→   457→            <div className="min-w-0">
   458→   458→              <p className="font-semibold">This paste has been flagged</p>
   459→   459→              <p className="mt-0.5 text-sm opacity-90">
   460→   460→                It was flagged by a moderator and may violate community guidelines.
   461→   461→              </p>
   462→   462→            </div>
   463→   463→          </div>
   464→   464→        </Reveal>
   465→   465→      )}
   466→   466→
   467→   467→      <div className="grid gap-6 lg:grid-cols-3">
   468→   468→        {/* ===== Main column ===== */}
   469→   469→        <div className="min-w-0 lg:col-span-2">
   470→   470→          {/* Paste header */}
   471→   471→          <Reveal variant={fadeUp} className="mb-4">
   472→   472→            <div className="flex flex-wrap items-start gap-3">
   473→   473→              <h1 className="min-w-0 flex-1 text-2xl font-bold tracking-tight sm:text-3xl">
   474→   474→                <span className="text-gradient">
   475→   475→                  {paste.title ||
   476→   476→                    (paste.contentType === 'file' ? paste.originalFileName : `/${paste.slug}`)}
   477→   477→                </span>
   478→   478→              </h1>
