    58→// Deduped language list — "Auto-detect" (plaintext) first.
    59→const LANGUAGE_OPTIONS: { value: string; label: string }[] = (() => {
    60→  const seen = new Set<string>(['plaintext'])
    61→  const out: { value: string; label: string }[] = [
    62→    { value: 'plaintext', label: 'Auto-detect' },
    63→  ]
    64→  for (const l of LANGUAGES) {
    65→    if (!seen.has(l.value)) {
    66→      out.push(l)
    67→      seen.add(l.value)
    68→    }
    69→  }
    70→  return out

   574→    setPasswordEnabled,
   575→    password,
   576→    setPassword,
   577→    burnAfterRead,
   578→    setBurnAfterRead,
   579→    expiryPreset,
   580→    setExpiryPreset,
   581→    expiryCustom,
   582→    setExpiryCustom,
   583→    submitLabel,
   584→    onSubmit,
   585→    submitting,
   586→    submitDisabled,
   587→    submitIcon,
   588→    hideLanguage,
   589→    onLogin,
   590→  } = props
   591→
   592→  return (
   593→    <div className="surface-2 space-y-4 rounded-xl p-4">
   594→      {/* Language */}
   595→      {!hideLanguage && language !== undefined && setLanguage && (
   596→        <div className="space-y-1.5">
   597→          <Label className={SECTION_LABEL_CLS}>Language</Label>
   598→          <Select value={language} onValueChange={setLanguage}>
   599→            <SelectTrigger className="w-full border-border bg-surface-3">
   600→              <SelectValue />
   601→            </SelectTrigger>
   602→            <SelectContent>
   603→              {LANGUAGE_OPTIONS.map((opt) => (
   604→                <SelectItem key={`${opt.value}-${opt.label}`} value={opt.value}>
   605→                  {opt.label}
   606→                </SelectItem>
   607→              ))}
   608→            </SelectContent>
   609→          </Select>
   610→        </div>
   611→      )}
   612→
   613→      {/* Visibility — 3 stacked radio cards */}
   614→      <div className="space-y-1.5">
   615→        <Label className={SECTION_LABEL_CLS}>Visibility</Label>
   616→        <div className="space-y-1.5">
   617→          {VISIBILITY_OPTIONS.map((opt) => {
   618→            const disabled = opt.value === 'private' && !user
   619→            const selected = visibility === opt.value
   620→            return (
   621→              <button
   622→                key={opt.value}
   623→                type="button"
   624→                disabled={disabled}
   625→                onClick={() => !disabled && setVisibility(opt.value)}
   626→                aria-pressed={selected}
   627→                className={[
   628→                  'flex w-full items-center gap-2 rounded-md border px-3 py-2 text-sm transition-colors',
   629→                  selected
   630→                    ? 'border-brand bg-brand/5 text-foreground'
   631→                    : 'border-border text-muted-foreground hover:text-foreground',
   632→                  disabled ? 'cursor-not-allowed opacity-50' : 'cursor-pointer',
   633→                ].join(' ')}
   634→              >
   635→                <opt.icon
   636→                  className={[
   637→                    'h-3.5 w-3.5',
   638→                    selected ? 'text-brand' : 'text-muted-foreground',
   639→                  ].join(' ')}
   640→                />
   641→                <span className="flex-1 text-left">{opt.label}</span>
   642→                {disabled && (
   643→                  <span className="font-mono text-[9px] uppercase tracking-wider text-muted-foreground">
   644→                    login required
   645→                  </span>
   646→                )}
   647→              </button>
   648→            )
   649→          })}
   650→        </div>
   651→      </div>
   652→
   653→      {/* Password */}
   654→      <div className="space-y-2">
   655→        <div className="flex items-center justify-between">
   656→          <Label
   657→            htmlFor="pw-switch"
   658→            className="flex items-center gap-1.5 text-sm font-medium"
   659→          >
   660→            <Lock className="h-3.5 w-3.5 text-muted-foreground" /> Password
   661→          </Label>
   662→          <Switch
   663→            id="pw-switch"
   664→            checked={passwordEnabled}
   665→            onCheckedChange={setPasswordEnabled}
   666→            disabled={burnAfterRead}
   667→            className="data-[state=checked]:bg-brand"
   668→          />
   669→        </div>
   670→        {passwordEnabled && (
   671→          <Input
   672→            type="password"
   673→            value={password}
   674→            onChange={(e) => setPassword(e.target.value)}
   675→            placeholder="Password"
   676→            className="bg-surface-3"
   677→            autoComplete="new-password"
   678→          />
   679→        )}
   680→      </div>
   681→
   682→      {/* Burn after read */}
   683→      <div className="space-y-1">
   684→        <div className="flex items-center justify-between">
   685→          <Label
   686→            htmlFor="burn-switch"
   687→            className="flex items-center gap-1.5 text-sm font-medium"
   688→          >
   689→            <Flame className="h-3.5 w-3.5 text-muted-foreground" /> Burn after
   690→            read
   691→          </Label>
   692→          <Switch
   693→            id="burn-switch"
   694→            checked={burnAfterRead}
   695→            onCheckedChange={setBurnAfterRead}
   696→            disabled={passwordEnabled}
   697→            className="data-[state=checked]:bg-brand-2"
   698→          />
   699→        </div>
   700→        <p className="text-[11px] text-muted-foreground">
   701→          Deleted after first view
   702→        </p>
   703→      </div>
   704→
   705→      {/* Expiry */}
   706→      <div className="space-y-1.5">
   707→        <Label htmlFor="expiry-select" className={SECTION_LABEL_CLS}>
   708→          Expires
   709→        </Label>
   710→        <Select value={expiryPreset} onValueChange={setExpiryPreset}>
   711→          <SelectTrigger
   712→            id="expiry-select"
   713→            className="w-full border-border bg-surface-3"
   714→          >
   715→            <SelectValue />
   716→          </SelectTrigger>
   717→          <SelectContent>
   718→            {EXPIRY_OPTIONS.map((opt) => (
   719→              <SelectItem key={opt.value} value={opt.value}>
   720→                {opt.label}
   721→              </SelectItem>
   722→            ))}
   723→          </SelectContent>
   724→        </Select>
   725→        {expiryPreset === 'custom' && (
   726→          <Input
   727→            type="datetime-local"
   728→            value={expiryCustom}
   729→            onChange={(e) => setExpiryCustom(e.target.value)}
   730→            className="bg-surface-3"
   731→            aria-label="Custom expiry date and time"
   732→          />
   733→        )}
   734→      </div>
   735→
   736→      {/* Submit */}
   737→      <Button
   738→        type="button"
   739→        onClick={onSubmit}
   740→        disabled={submitDisabled || submitting}
   741→        className="btn-shine relative w-full gap-1.5 overflow-hidden rounded-md bg-brand text-brand-foreground font-medium hover:bg-brand/90"
   742→        size="lg"
   743→      >
   744→        {submitting ? (
   745→          <Loader2 className="h-4 w-4 animate-spin" />
   746→        ) : (
   747→          submitIcon
   748→        )}
   749→        {submitLabel}
   750→      </Button>
   751→
   752→      {!user && (
   753→        <p className="text-center text-[11px] text-muted-foreground">
   754→          Guest paste —{' '}
   755→          <button
   756→            type="button"
   757→            onClick={onLogin}
   758→            className="font-medium text-brand hover:underline"
   759→          >
   760→            log in
   761→          </button>{' '}
   762→          for private pastes
   763→        </p>
   764→      )}
   765→    </div>
   766→  )
   767→}
