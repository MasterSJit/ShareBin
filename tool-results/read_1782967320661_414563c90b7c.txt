   496→function UsersTab() {
   497→  const me = useAppStore((s) => s.user)!
   498→  const [rows, setRows] = useState<UserRow[]>([])
   499→  const [loading, setLoading] = useState(true)
   500→  const [q, setQ] = useState('')
   501→  const [status, setStatus] = useState<string>('all')
   502→  const [confirm, setConfirm] = useState<{
   503→    user: UserRow
   504→    action: 'delete' | 'ban' | 'suspend'
   505→  } | null>(null)
   506→  const [busy, setBusy] = useState(false)
   507→
   508→  async function fetchUsers() {
   509→    setLoading(true)
   510→    try {
   511→      const r = await api.admin.listUsers(
   512→        q || undefined,
   513→        status === 'all' ? undefined : status,
   514→      )
   515→      setRows(r)
   516→    } catch (err) {
   517→      toast.error(err instanceof ApiError ? err.message : 'Failed to load users.')
   518→    } finally {
   519→      setLoading(false)
   520→    }
   521→  }
   522→
   523→  useEffect(() => {
   524→    fetchUsers()
   525→  }, [status])
   526→
   527→  async function patchUser(
   528→    u: UserRow,
   529→    patch: { status?: string; role?: string; verify?: boolean },
   530→  ) {
   531→    if (u.id === me.id && (patch.status === 'banned' || patch.status === 'suspended')) {
   532→      toast.error('You cannot suspend or ban yourself.')
   533→      return
   534→    }
   535→    try {
   536→      const res = await api.admin.updateUser(u.id, patch)
   537→      const updated = res[0] as UserRow | undefined
   538→      if (updated) {
   539→        setRows((cur) => cur.map((x) => (x.id === u.id ? { ...x, ...updated } : x)))
   540→        toast.success('User updated')
   541→      }
   542→    } catch (err) {
   543→      toast.error(err instanceof ApiError ? err.message : 'Update failed.')
   544→    }
   545→  }
   546→
   547→  async function doConfirm() {
   548→    if (!confirm) return
   549→    setBusy(true)
   550→    try {
   551→      if (confirm.action === 'delete') {
   552→        if (confirm.user.id === me.id) {
   553→          toast.error('You cannot delete yourself.')
   554→          return
   555→        }
   556→        await api.admin.deleteUser(confirm.user.id)
   557→        setRows((cur) => cur.filter((x) => x.id !== confirm.user.id))
   558→        toast.success('User deleted')
   559→      } else if (confirm.action === 'ban') {
   560→        if (confirm.user.id === me.id) {
   561→          toast.error('You cannot ban yourself.')
   562→          return
   563→        }
   564→        const res = await api.admin.updateUser(confirm.user.id, { status: 'banned' })
   565→        const updated = res[0] as UserRow | undefined
   566→        if (updated)
   567→          setRows((cur) =>
   568→            cur.map((x) => (x.id === confirm.user.id ? { ...x, ...updated } : x)),
   569→          )
   570→        toast.success('User banned')
   571→      } else if (confirm.action === 'suspend') {
   572→        if (confirm.user.id === me.id) {
   573→          toast.error('You cannot suspend yourself.')
   574→          return
   575→        }
   576→        const res = await api.admin.updateUser(confirm.user.id, { status: 'suspended' })
   577→        const updated = res[0] as UserRow | undefined
   578→        if (updated)
   579→          setRows((cur) =>
   580→            cur.map((x) => (x.id === confirm.user.id ? { ...x, ...updated } : x)),
   581→          )
   582→        toast.success('User suspended')
   583→      }
   584→    } catch (err) {
   585→      toast.error(err instanceof ApiError ? err.message : 'Action failed.')
   586→    } finally {
   587→      setBusy(false)
   588→      setConfirm(null)
   589→    }
   590→  }
   591→
   592→  return (
   593→    <div className="space-y-4">
   594→      {/* Filter bar */}
   595→      <Reveal variant={fadeUp}>
   596→        <div className="surface flex flex-wrap items-center gap-2 rounded-xl p-3">
