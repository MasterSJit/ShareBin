   242→      const msg = err instanceof ApiError ? err.message : 'Failed to create paste'
   243→      toast.error(msg)
   244→    } finally {
   245→      setSubmitting(false)
   246→    }
   247→  }

   249→  async function handleUploadFile() {

   251→      toast.error('Select a file first')
   252→      return
   253→    }
   254→    if (file.size > MAX_FILE_BYTES) {
   255→      toast.error(`File too large (max ${formatBytes(MAX_FILE_BYTES)})`)
   256→      return
   257→    }
   258→    if (visibility === 'private' && !user) {
   259→      toast.error('Log in to create private pastes')
   260→      return
   261→    }
   262→    setSubmitting(true)
   263→    try {
   264→      const fd = new FormData()
   265→      fd.append('file', file)
   266→      if (title.trim()) fd.append('title', title.trim())
   267→      if (description.trim()) fd.append('description', description.trim())
   268→      if (tags.trim()) fd.append('tags', tags.trim())
   269→      fd.append('visibility', visibility)
   270→      if (passwordEnabled && password) fd.append('password', password)
   271→      if (burnAfterRead) fd.append('burnAfterRead', 'true')
   272→      fd.append('expiryPreset', expiryPreset)
   273→      if (expiryPreset === 'custom' && expiryCustom) fd.append('expiryCustom', new Date(expiryCustom).toISOString())

   279→    } catch (err) {
   280→      const msg = err instanceof ApiError ? err.message : 'Failed to upload file'
   281→      toast.error(msg)
   282→    } finally {
   283→      setSubmitting(false)
   284→    }

   679→                      type="file"
   680→                      className="hidden"

   685→                        initial={{ opacity: 0, scale: 0.96 }}
   686→                        animate={{ opacity: 1, scale: 1 }}

   698→                          </p>
   699→                        </div>
   700→                        <button
   701→                          type="button"
   702→                          onClick={(e) => {

   705→                          }}

   713→                      <>
   714→                        <div className="flex h-12 w-12 items-center justify-center rounded-full bg-surface-3 text-muted-foreground">
   715→                          <Upload className="h-5 w-5" />
   716→                        </div>
   717→                        <div className="space-y-1">
   718→                          <p className="text-sm font-medium">Drop a file here</p>
   719→                          <p className="text-xs text-muted-foreground">or click to browse</p>

   725→                    )}
   726→                  </div>
   727→                </div>
   728→
   729→                {/* Sidebar */}

   737→                    submitIcon={<Upload className="h-4 w-4" />}
   738→                  />
   739→                </div>
   740→              </div>
   741→            </TabsContent>
   742→          </Tabs>
   743→        </div>
   744→      </div>
   745→    </div>
   746→  )
   747→}

   753→type SidebarCardProps = {
   754→  user: SafeUser | null
   755→  language?: string
   756→  setLanguage?: (v: string) => void
   757→  visibility: 'public' | 'unlisted' | 'private'
   758→  setVisibility: (v: 'public' | 'unlisted' | 'private') => void
   759→  passwordEnabled: boolean
   760→  setPasswordEnabled: (v: boolean) => void
   761→  password: string
   762→  setPassword: (v: string) => void
   763→  burnAfterRead: boolean
   764→  setBurnAfterRead: (v: boolean) => void
   765→  expiryPreset: string
   766→  setExpiryPreset: (v: string) => void
   767→  expiryCustom: string
   768→  setExpiryCustom: (v: string) => void
   769→  submitLabel: string
   770→  onSubmit: () => void
   771→  submitting: boolean
   772→  submitDisabled: boolean
   773→  submitIcon: React.ReactNode
   774→  hideLanguage?: boolean
   775→  onLogin: () => void
   776→  limits: {
   777→    name: string
   778→    allowedExpiryPresets: string[]
   779→    allowCustomExpiry: boolean
   780→    allowBurnAfterRead: boolean
   781→    allowPassword: boolean
   782→    allowPrivate: boolean
   783→    maxPasteSize: number
   784→    maxPastesPerMonth: number
   785→  } | null
   786→}
   787→
   788→function SidebarCard(props: SidebarCardProps) {
   789→  const {
   790→    user,
   791→    language,
   792→    setLanguage,
   793→    visibility,
   794→    setVisibility,
   795→    passwordEnabled,
   796→    setPasswordEnabled,
   797→    password,
   798→    setPassword,
   799→    burnAfterRead,
   800→    setBurnAfterRead,
   801→    expiryPreset,
   802→    setExpiryPreset,
   803→    expiryCustom,
   804→    setExpiryCustom,
   805→    submitLabel,
   806→    onSubmit,
   807→    submitting,
   808→    submitDisabled,
   809→    submitIcon,
   810→    hideLanguage,
   811→    onLogin,
   812→    limits,
   813→  } = props
   814→
   815→  // Magnetic button effect
   816→  const btnRef = useGsapMagnetic<HTMLButtonElement>(0.25)
   817→
   818→  // Gating flags (based on plan limits)
   819→  const canPrivate = limits ? limits.allowPrivate : false
   820→  const canBurn = limits ? limits.allowBurnAfterRead : false
   821→  const canPassword = limits ? limits.allowPassword : false
   822→  const canCustomExpiry = limits ? limits.allowCustomExpiry : false
   823→  const allowedPresets = limits ? limits.allowedExpiryPresets : ['never', '10m', '1h', '1d', '1w', '1M', '1y']
   824→  const planName = limits?.name || '—'
   825→
   826→  // If current selection is not allowed by plan, reset to 'never'
   827→  useEffect(() => {
   828→    if (limits && !allowedPresets.includes(expiryPreset) && !(expiryPreset === 'custom' && canCustomExpiry)) {
   829→      setExpiryPreset('never')
   830→    }
   831→    if (limits && !canBurn && burnAfterRead) setBurnAfterRead(false)
   832→    if (limits && !canPassword && passwordEnabled) setPasswordEnabled(false)
   833→    if (limits && !canPrivate && visibility === 'private') setVisibility('public')
   834→  }, [limits])
   835→
   836→  return (
   837→    <div className="surface-2 space-y-4 rounded-xl p-4">
   838→      {/* Plan badge */}
   839→      {limits && (
   840→        <div className="flex items-center justify-between rounded-md border border-border/60 bg-surface-3 px-2.5 py-1.5">
   841→          <span className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground">plan</span>
   842→          <span className="text-xs font-medium text-brand">{planName}</span>
   843→        </div>
   844→      )}
   845→
   846→      {/* Language */}
   847→      {!hideLanguage && language !== undefined && setLanguage && (
   848→        <div className="space-y-1.5">
   849→          <Label className={SECTION_LABEL_CLS}>Language</Label>
   850→          <Select value={language} onValueChange={setLanguage}>
   851→            <SelectTrigger className="w-full border-border bg-surface-3">
   852→              <SelectValue />
   853→            </SelectTrigger>
   854→            <SelectContent>
   855→              {LANGUAGE_OPTIONS.map((opt) => (
   856→                <SelectItem key={`${opt.value}-${opt.label}`} value={opt.value}>
   857→                  {opt.label}
   858→                </SelectItem>
   859→              ))}
   860→            </SelectContent>
   861→          </Select>
   862→        </div>
   863→      )}
   864→
   865→      {/* Visibility — 3 stacked radio cards */}
   866→      <div className="space-y-1.5">
   867→        <Label className={SECTION_LABEL_CLS}>Visibility</Label>
   868→        <div className="space-y-1.5">
   869→          {VISIBILITY_OPTIONS.map((opt) => {
   870→            const disabled = opt.value === 'private' && (!user || !canPrivate)
   871→            const selected = visibility === opt.value
   872→            return (
   873→              <button
   874→                key={opt.value}
   875→                type="button"
   876→                disabled={disabled}
   877→                onClick={() => !disabled && setVisibility(opt.value)}
   878→                aria-pressed={selected}
   879→                className={[
   880→                  'flex w-full items-center gap-2 rounded-md border px-3 py-2 text-sm transition-all',
   881→                  selected
   882→                    ? 'border-brand bg-brand/5 text-foreground'
   883→                    : 'border-border text-muted-foreground hover:text-foreground hover:border-border/60',
   884→                  disabled ? 'cursor-not-allowed opacity-50' : 'cursor-pointer',
   885→                ].join(' ')}
   886→              >
   887→                <opt.icon
   888→                  className={[
   889→                    'h-3.5 w-3.5 transition-colors',
   890→                    selected ? 'text-brand' : 'text-muted-foreground',
   891→                  ].join(' ')}
   892→                />
   893→                <span className="flex-1 text-left">{opt.label}</span>
   894→                {disabled && (
   895→                  <span className="font-mono text-[9px] uppercase tracking-wider text-muted-foreground">
   896→                    {!user ? 'login' : 'upgrade'}
   897→                  </span>
   898→                )}
   899→              </button>
   900→            )
   901→          })}
   902→        </div>
   903→      </div>
   904→
   905→      {/* Password */}
   906→      <div className="space-y-2">
   907→        <div className="flex items-center justify-between">
   908→          <Label
   909→            htmlFor="pw-switch"
   910→            className="flex items-center gap-1.5 text-sm font-medium"
   911→          >
   912→            <Lock className="h-3.5 w-3.5 text-muted-foreground" /> Password
   913→          </Label>
   914→          <Switch
   915→            id="pw-switch"
   916→            checked={passwordEnabled}
   917→            onCheckedChange={setPasswordEnabled}
   918→            disabled={burnAfterRead || !canPassword}
   919→            className="data-[state=checked]:bg-brand"
   920→          />
   921→        </div>
   922→        {!canPassword && (
   923→          <p className="font-mono text-[10px] text-muted-foreground">Requires higher plan</p>
   924→        )}
   925→        {passwordEnabled && (
   926→          <div className="field">
   927→            <input
   928→              type="password"
   929→              value={password}
   930→              onChange={(e) => setPassword(e.target.value)}
   931→              placeholder=" "
   932→              className="field-input"
   933→              id="paste-password"
   934→              autoComplete="new-password"
   935→            />
   936→            <label htmlFor="paste-password" className="field-float-label">
   937→              Password
   938→            </label>
   939→            <div className="field-bar" />
   940→          </div>
   941→        )}
   942→      </div>
   943→
   944→      {/* Burn after read */}
   945→      <div className="space-y-1">
   946→        <div className="flex items-center justify-between">
   947→          <Label
   948→            htmlFor="burn-switch"
   949→            className="flex items-center gap-1.5 text-sm font-medium"
   950→          >
   951→            <Flame className="h-3.5 w-3.5 text-muted-foreground" /> Burn after read
   952→          </Label>
   953→          <Switch
   954→            id="burn-switch"
   955→            checked={burnAfterRead}
   956→            onCheckedChange={setBurnAfterRead}
   957→            disabled={passwordEnabled || !canBurn}
   958→            className="data-[state=checked]:bg-brand-2"
   959→          />
   960→        </div>
   961→        <p className="text-[11px] text-muted-foreground">
   962→          {canBurn ? 'Deleted after first view' : 'Requires higher plan'}
   963→        </p>
   964→      </div>
   965→
   966→      {/* Expiry */}
   967→      <div className="space-y-1.5">
   968→        <Label htmlFor="expiry-select" className={SECTION_LABEL_CLS}>
   969→          Expires
   970→        </Label>
   971→        <Select value={expiryPreset} onValueChange={setExpiryPreset}>
   972→          <SelectTrigger
   973→            id="expiry-select"
   974→            className="w-full border-border bg-surface-3"
   975→          >
   976→            <SelectValue />
   977→          </SelectTrigger>
   978→          <SelectContent>
   979→            {EXPIRY_OPTIONS.map((opt) => {
   980→              const isCustom = opt.value === 'custom'
   981→              const isAllowed = isCustom ? canCustomExpiry : allowedPresets.includes(opt.value)
   982→              return (
   983→                <SelectItem
   984→                  key={opt.value}
   985→                  value={opt.value}
   986→                  disabled={!isAllowed}
   987→                  className={!isAllowed ? 'opacity-50' : ''}

   989→                  {opt.label}
   990→                  {!isAllowed && <span className="ml-auto font-mono text-[9px] text-muted-foreground">upgrade</span>}
   991→                </SelectItem>
   992→              )
   993→            })}
   994→          </SelectContent>
   995→        </Select>
   996→        {expiryPreset === 'custom' && (
   997→          <Input
   998→            type="datetime-local"
   999→            value={expiryCustom}
  1000→            onChange={(e) => setExpiryCustom(e.target.value)}
  1001→            className="bg-surface-3"
  1002→            aria-label="Custom expiry date and time"
  1003→          />
  1004→        )}
  1005→      </div>
  1006→
  1007→      {/* Submit — magnetic button */}
  1008→      <Button
  1009→        ref={btnRef}
  1010→        type="button"
  1011→        onClick={onSubmit}
  1012→        disabled={submitDisabled || submitting}

  1014→        size="lg"
  1015→      >
  1016→        {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : submitIcon}
  1017→        {submitLabel}
  1018→      </Button>
  1019→
  1020→      {!user && (
  1021→        <p className="text-center text-[11px] text-muted-foreground">
  1022→          Guest paste —{' '}
  1023→          <button
  1024→            type="button"
  1025→            onClick={onLogin}
  1026→            className="font-medium text-brand hover:underline"
  1027→          >
  1028→            log in
  1029→          </button>{' '}
  1030→          for more features
  1031→        </p>
  1032→      )}
  1033→    </div>
  1034→  )
  1035→}
